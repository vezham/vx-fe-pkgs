## Usage

Render this template in your server email callback. Pass your application branding and the server-generated action URL or verification code.

<ComponentPreview name="react-auth-email-email-changed-email">

```tsx title="email-changed-email.tsx"
// vx-bot/INFO: Adapted from Better Auth UI; see the Auth demos LICENSE.
import { AuthEmail } from '@vezham/react/auth/email'

import { EmailPreview } from '../email-preview'

export const EmailChangedEmailDemo = () => (
  <EmailPreview title="Email Changed Email Preview">
    <AuthEmail.EmailChanged
      oldEmail="old@example.com"
      newEmail="new@example.com"
      appName="Vezham"
      supportEmail="support@example.com"
      revertURL="https://example.com/revert-email"
      darkMode={true}
      poweredBy={true}
    />
  </EmailPreview>
)
```

```tsx title="email-preview.tsx"
import { type ReactNode, useEffect, useState } from 'react'

type Props = { children: ReactNode; title: string }

export const EmailPreview = ({ children, title }: Props) => {
  const [html, setHtml] = useState<string>()
  const [failed, setFailed] = useState(false)

  useEffect(() => {
    let active = true
    void (async () => {
      try {
        const { renderToReadableStream } = await import('react-dom/server')
        const stream = await renderToReadableStream(children)
        const value = await new Response(stream).text()
        if (active) setHtml(value)
      } catch {
        if (active) setFailed(true)
      }
    })()
    return () => {
      active = false
    }
  }, [children])

  if (failed) return <p role="alert">The email preview could not render.</p>
  if (!html) return <p role="status">Rendering email…</p>
  return (
    <iframe
      title={title}
      srcDoc={html}
      sandbox=""
      className="h-[680px] w-full rounded-lg border bg-white"
    />
  )
}
```

## Props

```tsx
import type { EmailChangedEmailProps } from '@vezham/react/auth/email'
```

| Prop           | Type                                        | Required | Description                                                                         |
| -------------- | ------------------------------------------- | -------- | ----------------------------------------------------------------------------------- |
| `oldEmail`     | `string`                                    | No       | The previous email address that was changed                                         |
| `newEmail`     | `string`                                    | No       | The new email address                                                               |
| `revertURL`    | `string`                                    | No       | URL to revert the email change if unauthorized                                      |
| `appName`      | `string`                                    | No       | Name of the application sending the email                                           |
| `supportEmail` | `string`                                    | No       | Support email address for security concerns                                         |
| `logoURL`      | `string \| { light: string; dark: string }` | No       | Logo URL(s) - a single string or light/dark variants. If omitted, no logo is shown. |
| `classNames`   | `EmailClassNames`                           | No       | Custom CSS class names for styling specific parts of the email                      |
| `colors`       | `EmailColors`                               | No       | Custom color scheme for light and dark modes                                        |
| `poweredBy`    | `boolean`                                   | No       | Whether to show the "Powered by better-auth" footer                                 |
| `darkMode`     | `boolean`                                   | No       | Whether to enable dark mode support                                                 |
| `head`         | `ReactNode`                                 | No       | Additional React nodes to inject into the email head                                |
| `localization` | `Partial<EmailChangedEmailLocalization>`    | No       | Localization overrides for customizing email text                                   |

Use the exported props type for inherited configuration and component props.
