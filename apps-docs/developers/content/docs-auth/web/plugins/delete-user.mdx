The delete-user plugin renders the UI for Better Auth's built-in [account deletion](https://www.better-auth.com/docs/concepts/users-accounts#delete-user) feature. Users can permanently delete their account with a confirmation dialog via the `<DeleteUser />` card, wrapped by `<DeleteUser.DangerZone />` in security settings.

## Setup

<Steps>

<Step>
### Enable account deletion in Better Auth

[Better Auth core includes account deletion](https://www.better-auth.com/docs/concepts/users-accounts#delete-user). You do not need another plugin.

Enable the feature by setting `user.deleteUser.enabled` to `true`:

```ts title="lib/auth.ts"
import { betterAuth } from '@vezham/auth'

export const auth = betterAuth({
  // ...
  user: {
    deleteUser: {
      // [!code highlight]
      enabled: true // [!code highlight]
    } // [!code highlight]
  }
})
```

After you enable the feature, the client provides `authClient.deleteUser()`. You do not need a client plugin.

For an OAuth user without a password, provide `sendDeleteAccountVerification`. This callback lets the user confirm deletion by email.

</Step>

<Step>

### Register the UI plugin

Pass `DeleteUser.plugin()` to `<Auth.Provider>`:

```tsx title="components/providers.tsx"
import { Auth } from '@vezham/react/auth'
import { DeleteUser } from '@vezham/react/auth/plugins/delete-user'

// [!code highlight]
;<Auth.Provider
  authClient={authClient}
  navigate={navigate}
  plugins={[DeleteUser.plugin()]} // [!code highlight]
>
  {children}
</Auth.Provider>
```

</Step>

</Steps>

```tsx title="demo-provider.tsx"
import { QueryClient } from '@tanstack/react-query'
import {
  type ReactNode,
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState
} from 'react'

import { Description } from '@vezham/react-v3'
import { Auth, type AuthProviderProps } from '@vezham/react/auth'

import { type PreviewClient, createPreviewClient } from './demo-client'

type DemoContextValue = {
  authClient: PreviewClient
  navigate: AuthProviderProps['navigate']
  theme: string
  setTheme: (theme: string) => void
}

const DemoContext = createContext<DemoContextValue | null>(null)

export const DemoProvider = ({
  children,
  authenticated = false
}: {
  children: ReactNode
  authenticated?: boolean
}) => {
  const [authClient] = useState(() => createPreviewClient(authenticated))
  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: { retry: false },
          mutations: { retry: false }
        }
      })
  )
  const [destination, setDestination] = useState<string | null>(null)
  const [theme, setTheme] = useState('system')
  const navigate = useCallback<AuthProviderProps['navigate']>(({ to }) => {
    if (to.includes('/auth/') || to.includes('/settings')) setDestination(to)
  }, [])
  const segment = destination
    ?.split(/[?#]/, 1)[0]
    .split('/')
    .filter(Boolean)
    .pop()
  useEffect(() => () => queryClient.clear(), [queryClient])

  return (
    <DemoContext.Provider value={{ authClient, navigate, theme, setTheme }}>
      <div
        className="flex w-full max-w-xl flex-col items-center gap-5"
        onClickCapture={event => {
          const target = event.target
          if (!(target instanceof Element)) return
          const href = target.closest('a')?.getAttribute('href')
          if (href === '#') event.preventDefault()
          else if (
            href?.startsWith('/auth/') ||
            href?.startsWith('/settings')
          ) {
            event.preventDefault()
            navigate({ to: href })
          }
        }}>
        <Description className="text-center">
          Local preview. Use ada@example.com to sign in. Account changes are
          simulated.
        </Description>
        <Auth.Provider
          authClient={authClient}
          queryClient={queryClient}
          navigate={navigate}
          socialProviders={['github', 'google']}
          redirectTo="/settings/security">
          {destination?.includes('/settings') ? (
            <Auth.Settings path={segment ?? 'account'} />
          ) : destination ? (
            <Auth path={segment ?? 'sign-in'} />
          ) : (
            children
          )}
        </Auth.Provider>
      </div>
    </DemoContext.Provider>
  )
}

export const useDemo = () => {
  const value = useContext(DemoContext)
  if (!value) throw new Error('Auth examples require DemoProvider')
  return value
}

export const useDemoTheme = () => {
  const { theme, setTheme } = useDemo()
  return { theme, setTheme, themes: ['system', 'light', 'dark'] }
}
```

```tsx title="demo-client.ts"
import { apiKeyClient } from '@vezham/auth-api-key/client'
import { passkeyClient } from '@vezham/auth-passkey/client'
import {
  deviceAuthorizationClient,
  magicLinkClient,
  multiSessionClient,
  organizationClient,
  usernameClient
} from '@vezham/auth/client/plugins'
import { type ReactAuthClient, createAuthClient } from '@vezham/auth/react'

export const demoUser = {
  id: 'auth-preview-user',
  name: 'Ada Lovelace',
  username: 'ada',
  displayUsername: 'ada',
  email: 'ada@example.com',
  emailVerified: true,
  image: null,
  createdAt: new Date('2026-01-01'),
  updatedAt: new Date('2026-01-01')
}

type PreviewPlugins = [
  ReturnType<typeof apiKeyClient>,
  ReturnType<typeof passkeyClient>,
  ReturnType<typeof organizationClient>,
  ReturnType<typeof usernameClient>,
  ReturnType<typeof magicLinkClient>,
  ReturnType<typeof multiSessionClient>,
  ReturnType<typeof deviceAuthorizationClient>
]

export type PreviewClient = ReactAuthClient<{ plugins: PreviewPlugins }>

export const createPreviewClient = (authenticated: boolean): PreviewClient => {
  let signedIn = authenticated
  const organization = {
    id: 'preview-organization',
    name: 'Analytical Engines',
    slug: 'analytical-engines',
    logo: null,
    createdAt: new Date('2026-01-01')
  }
  const member = {
    id: 'preview-member',
    organizationId: organization.id,
    userId: demoUser.id,
    role: 'owner',
    createdAt: new Date('2026-01-01'),
    user: demoUser
  }
  const session = {
    user: demoUser,
    session: {
      id: 'auth-preview-session',
      token: 'local-preview-token',
      userId: demoUser.id,
      activeOrganizationId: organization.id,
      expiresAt: new Date('2099-01-01'),
      createdAt: new Date('2026-01-01'),
      updatedAt: new Date('2026-01-01'),
      ipAddress: '127.0.0.1',
      userAgent: 'Vezham authentication preview'
    }
  }

  return createAuthClient<{
    plugins: PreviewPlugins
    baseURL: string
    fetchOptions: { customFetchImpl: typeof fetch }
  }>({
    baseURL: 'https://auth-preview.invalid',
    plugins: [
      apiKeyClient(),
      passkeyClient(),
      organizationClient({ teams: { enabled: true } }),
      usernameClient(),
      magicLinkClient(),
      multiSessionClient(),
      deviceAuthorizationClient()
    ],
    fetchOptions: {
      customFetchImpl: async (input, init) => {
        const url = new URL(
          input instanceof Request ? input.url : String(input)
        )
        const path = url.pathname
        if (path.endsWith('/get-session'))
          return Response.json(signedIn ? session : null)
        if (path.endsWith('/list-sessions'))
          return Response.json(signedIn ? [session.session] : [])
        if (path.endsWith('/list-accounts'))
          return Response.json([
            {
              id: 'credential',
              providerId: 'credential',
              accountId: demoUser.id
            }
          ])
        if (path.endsWith('/sign-out')) {
          signedIn = false
          return Response.json({ success: true })
        }
        if (
          path.endsWith('/sign-in/email') ||
          path.endsWith('/sign-in/username')
        ) {
          const values = JSON.parse(String(init?.body)) as {
            email?: string
            username?: string
          }
          if (
            values.email?.toLowerCase() === demoUser.email ||
            values.username === demoUser.username
          ) {
            signedIn = true
            return Response.json({
              token: session.session.token,
              user: demoUser
            })
          }
        }
        if (path.endsWith('/is-username-available'))
          return Response.json({ available: true })
        if (path.endsWith('/multi-session/list-device-sessions'))
          return Response.json([session])
        if (path.endsWith('/api-key/list'))
          return Response.json({ apiKeys: [], total: 0, limit: 10, offset: 0 })
        if (path.endsWith('/passkey/list-user-passkeys'))
          return Response.json([])
        if (path.endsWith('/organization/list'))
          return Response.json([organization])
        if (path.endsWith('/organization/get-full-organization'))
          return Response.json({
            ...organization,
            members: [member],
            invitations: [],
            teams: []
          })
        if (path.endsWith('/organization/get-active-member'))
          return Response.json(member)
        if (path.endsWith('/organization/get-active-member-role'))
          return Response.json({ role: member.role })
        if (path.endsWith('/organization/list-members'))
          return Response.json({ members: [member], total: 1 })
        if (path.endsWith('/organization/has-permission'))
          return Response.json({ success: true, error: null })
        if (
          /\/organization\/(list-invitations|list-user-invitations|list-teams|list-user-teams|list-organization-teams|list-roles)$/.test(
            path
          )
        )
          return Response.json([])
        return Response.json(
          {
            code: 'PREVIEW_VALIDATION',
            message:
              'Preview response: use ada@example.com to sign in. Account changes are disabled in this demo.'
          },
          { status: 400 }
        )
      }
    }
  })
}
```

The `<DeleteUser.DangerZone />` card is automatically rendered in `<Auth.Settings.Security />` when the plugin is registered. It renders a danger zone heading and the `<DeleteUser />` card below it.

**Usage**

```tsx
import { DeleteUser } from '@vezham/react/auth/plugins/delete-user'

;<DeleteUser.DangerZone />
```

**Props**

```tsx title="DangerZoneProps"
type DangerZoneProps = {
  className?: string
  variant?: CardProps['variant']
}
```

### `<DeleteUser />`

The delete account card with confirmation dialog. Used inside `<DeleteUser.DangerZone />` by default. Import it directly if you need a custom layout.

When credential confirmation is required, the password starts masked and includes a localized show/hide control.

**Usage**

```tsx
import { DeleteUser } from '@vezham/react/auth/plugins/delete-user'

;<DeleteUser />
```

**Props**

```tsx title="DeleteAccountProps"
type DeleteAccountProps = {
  className?: string
  variant?: CardProps['variant']
}
```

## Options

```tsx title="DeleteUserPluginOptions"
type DeleteUserPluginOptions = {
  /**
   * When `true`, matches server `sendDeleteAccountVerification`: deletion starts by sending a
   * verification email instead of deleting immediately in this request.
   */
  sendDeleteAccountVerification?: boolean
  /**
   * Override the plugin's default localization strings.
   * @remarks `DeleteUserLocalization`
   */
  localization?: Partial<DeleteUserLocalization>
}
```
