The passkey plugin adds passwordless authentication using WebAuthn. Users can sign in with their device authenticator (Touch ID, Face ID, Windows Hello) and manage registered passkeys from their security settings.

It contributes:

- A "Continue with Passkey" button rendered on the sign-in and magic-link views (hidden on sign-up)
- A `<Passkey.Item />` security card for listing, adding, renaming, and deleting registered passkeys
- `useSignInPasskey`, `useAddPasskey`, `useUpdatePasskey`, `useDeletePasskey`, and `useListPasskeys` hooks

## Setup

<Steps>

<Step>

### Install the Better Auth plugin

Install the [`@vezham/auth-passkey`](https://www.better-auth.com/docs/plugins/passkey) package and add it to your Better Auth server config:

```ts title="lib/auth.ts"
import { betterAuth } from '@vezham/auth'
import { passkey } from '@vezham/auth-passkey'

// [!code highlight]

export const auth = betterAuth({
  // ...
  plugins: [
    passkey() // [!code highlight]
  ]
})
```

</Step>

<Step>

### Install the matching client plugin

Add `passkeyClient()` to your auth client so `authClient.signIn.passkey` and `authClient.passkey.*` are available:

```ts title="lib/auth-client.ts"
import { passkeyClient } from '@vezham/auth-passkey/client'
import { createAuthClient } from '@vezham/auth/react'

// [!code highlight]

export const authClient = createAuthClient({
  plugins: [passkeyClient()] // [!code highlight]
})
```

</Step>

<Step>

### Register the UI plugin

Pass `Passkey.plugin()` to `<Auth.Provider>`:

```tsx title="components/providers.tsx"
import { Auth } from '@vezham/react/auth'
import { Passkey } from '@vezham/react/auth/plugins/passkey'

// [!code highlight]
;<Auth.Provider
  authClient={authClient}
  navigate={navigate}
  plugins={[Passkey.plugin()]} // [!code highlight]
>
  {children}
</Auth.Provider>
```

</Step>

</Steps>

```tsx title="demo-provider.tsx"
import { QueryClient } from '@tanstack/react-query'
import {
  type ReactNode,
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState
} from 'react'

import { Description } from '@vezham/react-v3'
import { Auth, type AuthProviderProps } from '@vezham/react/auth'

import { type PreviewClient, createPreviewClient } from './demo-client'

type DemoContextValue = {
  authClient: PreviewClient
  navigate: AuthProviderProps['navigate']
  theme: string
  setTheme: (theme: string) => void
}

const DemoContext = createContext<DemoContextValue | null>(null)

export const DemoProvider = ({
  children,
  authenticated = false
}: {
  children: ReactNode
  authenticated?: boolean
}) => {
  const [authClient] = useState(() => createPreviewClient(authenticated))
  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: { retry: false },
          mutations: { retry: false }
        }
      })
  )
  const [destination, setDestination] = useState<string | null>(null)
  const [theme, setTheme] = useState('system')
  const navigate = useCallback<AuthProviderProps['navigate']>(({ to }) => {
    if (to.includes('/auth/') || to.includes('/settings')) setDestination(to)
  }, [])
  const segment = destination
    ?.split(/[?#]/, 1)[0]
    .split('/')
    .filter(Boolean)
    .pop()
  useEffect(() => () => queryClient.clear(), [queryClient])

  return (
    <DemoContext.Provider value={{ authClient, navigate, theme, setTheme }}>
      <div
        className="flex w-full max-w-xl flex-col items-center gap-5"
        onClickCapture={event => {
          const target = event.target
          if (!(target instanceof Element)) return
          const href = target.closest('a')?.getAttribute('href')
          if (href === '#') event.preventDefault()
          else if (
            href?.startsWith('/auth/') ||
            href?.startsWith('/settings')
          ) {
            event.preventDefault()
            navigate({ to: href })
          }
        }}>
        <Description className="text-center">
          Local preview. Use ada@example.com to sign in. Account changes are
          simulated.
        </Description>
        <Auth.Provider
          authClient={authClient}
          queryClient={queryClient}
          navigate={navigate}
          socialProviders={['github', 'google']}
          redirectTo="/settings/security">
          {destination?.includes('/settings') ? (
            <Auth.Settings path={segment ?? 'account'} />
          ) : destination ? (
            <Auth path={segment ?? 'sign-in'} />
          ) : (
            children
          )}
        </Auth.Provider>
      </div>
    </DemoContext.Provider>
  )
}

export const useDemo = () => {
  const value = useContext(DemoContext)
  if (!value) throw new Error('Auth examples require DemoProvider')
  return value
}

export const useDemoTheme = () => {
  const { theme, setTheme } = useDemo()
  return { theme, setTheme, themes: ['system', 'light', 'dark'] }
}
```

```tsx title="demo-client.ts"
import { passkeyClient } from '@vezham/auth-passkey/client'
import {

  magicLinkClient,
  multiSessionClient,

} from '@vezham/auth/client/plugins'
import { type ReactAuthClient, createAuthClient } from '@vezham/auth/react'

export const demoUser = {
  id: 'auth-preview-user',
  name: 'Ada Lovelace',
  username: 'ada',
  displayUsername: 'ada',
  email: 'ada@example.com',
  emailVerified: true,
  image: null,
  createdAt: new Date('2026-01-01'),
  updatedAt: new Date('2026-01-01')
}

type PreviewPlugins = [
  ReturnType<typeof apiKeyClient>,
  ReturnType<typeof passkeyClient>,
  ReturnType<typeof organizationClient>,
  ReturnType<typeof usernameClient>,
  ReturnType<typeof magicLinkClient>,
  ReturnType<typeof multiSessionClient>,
  ReturnType<typeof deviceAuthorizationClient>
]

export type PreviewClient = ReactAuthClient<{ plugins: PreviewPlugins }>

export const createPreviewClient = (authenticated: boolean): PreviewClient => {
  let signedIn = authenticated
  const organization = {
    id: 'preview-organization',
    name: 'Analytical Engines',
    slug: 'analytical-engines',
    logo: null,
    createdAt: new Date('2026-01-01')
  }
  const member = {
    id: 'preview-member',
    organizationId: organization.id,
    userId: demoUser.id,
    role: 'owner',
    createdAt: new Date('2026-01-01'),
    user: demoUser
  }
  const session = {
    user: demoUser,
    session: {
      id: 'auth-preview-session',
      token: 'local-preview-token',
      userId: demoUser.id,
      activeOrganizationId: organization.id,
      expiresAt: new Date('2099-01-01'),
      createdAt: new Date('2026-01-01'),
      updatedAt: new Date('2026-01-01'),
      ipAddress: '127.0.0.1',
      userAgent: 'Vezham authentication preview'
    }
  }

  return createAuthClient<{
    plugins: PreviewPlugins
    baseURL: string
    fetchOptions: { customFetchImpl: typeof fetch }
  }>({
    baseURL: 'https://auth-preview.invalid',
    plugins: [
      apiKeyClient(),
      passkeyClient(),
      organizationClient({ teams: { enabled: true } }),
      usernameClient(),
      magicLinkClient(),
      multiSessionClient(),
      deviceAuthorizationClient()
    ],
    fetchOptions: {
      customFetchImpl: async (input, init) => {
        const url = new URL(
          input instanceof Request ? input.url : String(input)
        )
        const path = url.pathname
        if (path.endsWith('/get-session'))
          return Response.json(signedIn ? session : null)
        if (path.endsWith('/list-sessions'))
          return Response.json(signedIn ? [session.session] : [])
        if (path.endsWith('/list-accounts'))
          return Response.json([
            {
              id: 'credential',
              providerId: 'credential',
              accountId: demoUser.id
            }
          ])
        if (path.endsWith('/sign-out')) {
          signedIn = false
          return Response.json({ success: true })
        }
        if (
          path.endsWith('/sign-in/email') ||
          path.endsWith('/sign-in/username')
        ) {
          const values = JSON.parse(String(init?.body)) as {
            email?: string
            username?: string
          }
          if (
            values.email?.toLowerCase() === demoUser.email ||
            values.username === demoUser.username
          ) {
            signedIn = true
            return Response.json({
              token: session.session.token,
              user: demoUser
            })
          }
        }
        if (path.endsWith('/is-username-available'))
          return Response.json({ available: true })
        if (path.endsWith('/multi-session/list-device-sessions'))
          return Response.json([session])
        if (path.endsWith('/api-key/list'))
          return Response.json({ apiKeys: [], total: 0, limit: 10, offset: 0 })
        if (path.endsWith('/passkey/list-user-passkeys'))
          return Response.json([])
        if (path.endsWith('/organization/list'))
          return Response.json([organization])
        if (path.endsWith('/organization/get-full-organization'))
          return Response.json({
            ...organization,
            members: [member],
            invitations: [],
            teams: []
          })
        if (path.endsWith('/organization/get-active-member'))
          return Response.json(member)
        if (path.endsWith('/organization/get-active-member-role'))
          return Response.json({ role: member.role })
        if (path.endsWith('/organization/list-members'))
          return Response.json({ members: [member], total: 1 })
        if (path.endsWith('/organization/has-permission'))
          return Response.json({ success: true, error: null })
        if (
          /\/organization\/(list-invitations|list-user-invitations|list-teams|list-user-teams|list-organization-teams|list-roles)$/.test(
            path
          )
        )
          return Response.json([])
        return Response.json(
          {
            code: 'PREVIEW_VALIDATION',
            message:
              'Preview response: use ada@example.com to sign in. Account changes are disabled in this demo.'
          },
          { status: 400 }
        )
      }
    }
  })
}
```

## Passkey registration policy

By default, the add-passkey dialog does not set `authenticatorAttachment`.
The browser and operating system show the available passkey options.

Set a preference in the plugin when all registrations must use one authenticator type:

```ts
Passkey.plugin({ authenticatorAttachment: 'platform' })
Passkey.plugin({ authenticatorAttachment: 'cross-platform' })
```

`"platform"` prefers the current device. `"cross-platform"` prefers a security key or another device. The dialog does not show an attachment selector.

`useAddPasskey` accepts every parameter exposed by `authClient.passkey.addPasskey`. Use this hook for custom registration flows:

```tsx
const { mutate: addPasskey } = useAddPasskey(authClient as PasskeyAuthClient)

addPasskey({
  name: 'Work laptop',
  authenticatorAttachment: 'platform',
  extensions: { credProps: true },
  returnWebAuthnResponse: true
})
```

<Callout>
  `residentKey` and `userVerification` are server plugin policies. They are not
  parameters of `authClient.passkey.addPasskey`, so the client UI does not
  expose them.
</Callout>

## Passkey autofill

With the plugin registered, the sign-in form asks the browser to offer saved
passkeys straight from its autofill dropdown, so most people never press the
passkey button at all. This is the WebAuthn conditional UI flow.

Two pieces make it work, and the built-in components already handle both:

- The identifier and password fields carry `webauthn` as the last token of
  their `autocomplete` attribute, added by `withPasskeyAutoFill`.
- `<Passkey.Button />` calls `usePasskeyAutoFill`, which opens a conditionally
  mediated request once the browser reports that it supports one.

Browsers without conditional mediation ignore the extra token and never get the
request, so the button stays as the fallback everywhere.

Turn the whole thing off with:

```ts
Passkey.plugin({ autoFill: false })
```

If you write your own sign-in form, add the token and start the conditional
request:

```tsx
import type { PasskeyAuthClient } from "@vezham/auth-ui-core/plugins/passkey"
import {
  isPasskeyAutoFillEnabled,
  withPasskeyAutoFill
} from "@vezham/auth-ui-core/plugins/passkey"
import { usePasskeyAutoFill } from "@vezham/auth-ui-react/plugins/passkey"

const { authClient, plugins } = useAuth()
const passkeyAutoFill = isPasskeyAutoFillEnabled(plugins)

usePasskeyAutoFill(authClient as PasskeyAuthClient)

<Input autoComplete={withPasskeyAutoFill("email", passkeyAutoFill)} />
```

<Callout type="warn">
`navigator.credentials.get()` accepts an `AbortSignal`. The bundled hook calls
the Better Auth passkey client, which does not expose that signal. Unmounting
the form only stops a pending availability probe. It does not cancel a request
that already started.

If your custom implementation calls `navigator.credentials.get()` directly,
pass an `AbortSignal` and abort it during cleanup.

</Callout>

## Options

```ts
Passkey.plugin({
  // Omit this option to let the browser show all available choices.
  authenticatorAttachment: 'platform',
  // Override any of the plugin's localization strings.
  localization: {
    passkeys: 'Security Keys'
  }
})
```

```tsx title="PasskeyPluginOptions"
type PasskeyPluginOptions = {
  /**
   * Offer passkeys through the browser's autofill dropdown (WebAuthn
   * conditional UI) as soon as a sign-in form mounts, instead of waiting for
   * the user to press the passkey button.
   *
   * Requires the `webauthn` autocomplete token on the identifier field, which
   * the built-in sign-in forms add for you.
   * @default true
   */
  autoFill?: boolean
  /**
   * Request one authenticator type for passkey registration. Omit this option
   * to let the browser and operating system show all available choices.
   *
   * `"platform"` prefers this device. `"cross-platform"` prefers a security
   * key or another device.
   */
  authenticatorAttachment?: PasskeyAuthenticatorAttachment
  /**
   * Override the plugin's default localization strings.
   * @remarks `PasskeyLocalization`
   */
  localization?: Partial<PasskeyLocalization>
}
```
